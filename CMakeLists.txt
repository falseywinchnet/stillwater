cmake_minimum_required(VERSION 3.22)
project(Stillwater VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STILLWATER_SANITIZERS "Enable address and undefined behavior checks" OFF)
add_library(stillwater_core src/scene.cpp src/habitat.cpp src/sound.cpp src/leaf_texture.cpp src/leaf_shadow.cpp)
target_include_directories(stillwater_core PUBLIC include)
target_compile_options(stillwater_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(STILLWATER_SANITIZERS)
  target_compile_options(stillwater_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(stillwater_core PUBLIC -fsanitize=address,undefined)
endif()
find_package(ZLIB REQUIRED)
target_link_libraries(stillwater_core PRIVATE ZLIB::ZLIB)
enable_testing()
add_executable(stillwater_tests tests/core_tests.cpp)
target_link_libraries(stillwater_tests PRIVATE stillwater_core)
target_compile_definitions(stillwater_tests PRIVATE STILLWATER_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_test(NAME scene_and_sound COMMAND stillwater_tests)
if(APPLE)
  add_executable(Stillwater MACOSX_BUNDLE src/main.cpp src/macos_host.cpp src/metal_renderer.cpp src/macos_audio.cpp src/renderer_verification.cpp)
  target_link_libraries(Stillwater PRIVATE stillwater_core "-framework AppKit" "-framework QuartzCore" "-framework Metal" "-framework CoreGraphics" "-framework ImageIO" "-framework AudioToolbox" "-framework CoreFoundation" objc)
  target_compile_options(Stillwater PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  set_target_properties(Stillwater PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/assets/Info.plist.in")
  add_custom_target(stillwater_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_BUNDLE_CONTENT_DIR:Stillwater>/Resources"
    DEPENDS Stillwater
    COMMENT "Synchronizing aquarium resources")
endif()
